\subsection{Identification in Causal Mediation}
\label{appendix:identification}
\citet[Theorem~1]{imai2010identification} states that the ADE and AIE are identified under sequential quasi-random assignment.
Write $F_{\vec X_i}$ for the cumulative density function for control variables $\vec X_i$.
For each $z = 0,1$: \\
\makebox[\textwidth]{\parbox{1.25\textwidth}{
\begin{align*}
    \E{ Y_i(1, D_i(z)) - Y_i(0, D_i(z))}
    &= \int \int 
    \Big( \Egiven{ Y_i }{Z_i = 1, D_i, \vec X_i}
        - \Egiven{ Y_i }{Z_i = 0, D_i, \vec X_i} \Big)
            dF_{D_i \, | \, Z_i = z, \vec X_i} dF_{\vec X_i}, \\
    \E{ Y_i(z, D_i(1)) - Y_i(z, D_i(0))}
    &= \int \int \Egiven{ Y_i }{Z_i = z, D_i, \vec X_i}
    \Big( dF_{D_i \, | \, Z_i = 1, \vec X_i}
        - dF_{D_i \, | \, Z_i = 0, \vec X_i} \Big) dF_{\vec X_i}.
\end{align*}
}}

In the  setting with $D_i =0,1$ binary considered in this paper, these are written as sums instead.
For each $z=0,1$,
\begin{align*}
    \operatorname{ADE}(z)
    &= \E{Y_i(1,D_i(z))-Y_i(0,D_i(z))} \\
    &= \int \sum_{d \in \{0,1\}} \Big(
        \Egiven{Y_i}{Z_i=1,D_i=d,\vec X_i=\vec x} - \Egiven{Y_i}{Z_i=0,D_i=d,\vec X_i} \Big) \\
    &\qquad\qquad\times \Prob{D_i=d\mid Z_i=z,\vec X_i} \,dF_{\vec X_i}.
\end{align*}
Likewise, for the average natural indirect effect, for each $z=0,1$
\begin{align*}
    \operatorname{AIE}(z)
    &= \E{Y_i(z,D_i(1))-Y_i(z,D_i(0))} \\
    &= \int \sum_{d \in \{0,1\}} \Egiven{Y_i}{Z_i=z,D_i=d,\vec X_i} \\
    &\qquad\qquad
    \times \Big(\Prob{D_i=d\mid Z_i=1,\vec X_i} - \Prob{D_i=d\mid Z_i=0,\vec X_i=\vec x} \Big) \,dF_{\vec X_i}.
\end{align*}

\subsection{Bias in Causal Mediation (CM) Estimands}
\label{appendix:mediation-bias}
Suppose that $Z_i$ is ignorable conditional on $\vec X_i$, but $D_i$ is not.


\subsubsection{Bias in the Average Direct Effect (ADE)}
Consider the conventional estimand for $\operatorname{ADE}(z)$, for each $z=0,1$.
\begin{align*}
    & \E[D_i \,\mid\, Z_i=z, \vec X_i]{
        \Egiven{Y_i}{Z_i=1,D_i,\vec X_i} -
        \Egiven{Y_i}{Z_i=0,D_i,\vec X_i} } \\
    &=
    \sum_{d \in \{0,1\}}
    \Probgiven{D_i=d}{Z_i=z,\vec X_i}
    \Big(
        \Egiven{Y_i}{Z_i=1,D_i=d,\vec X_i}
        -
        \Egiven{Y_i}{Z_i=0,D_i=d,\vec X_i}
    \Big).
\end{align*}
Conditional on $\vec X_i$, it averages the outcome contrast between $Z_i=1$ and $Z_i=0$ at each fixed mediator value $d$ over the mediator distribution under treatment state $z$.

Fix $d \in \{0,1\}$.
Treatment quasi-random assignment and consistency give,
\begin{align*}
    \Egiven{Y_i}{Z_i=1,D_i=d,\vec X_i}
    &=
    \Egiven{Y_i(1,d)}{D_i(1)=d,\vec X_i},
    \\
    \Egiven{Y_i}{Z_i=0,D_i=d,\vec X_i}
    &=
    \Egiven{Y_i(0,d)}{D_i(0)=d,\vec X_i}.
\end{align*}
Therefore,
\begin{align*}
    &\Egiven{Y_i}{Z_i=1,D_i=d,\vec X_i}
    -
    \Egiven{Y_i}{Z_i=0,D_i=d,\vec X_i}
    \\
    &=
    \Egiven{Y_i(1,d)}{D_i(1)=d,\vec X_i}
    -
    \Egiven{Y_i(0,d)}{D_i(0)=d,\vec X_i}.
\end{align*}

Add and subtract
$\Egiven{Y_i(0,d)}{D_i(1)=d,\vec X_i}$, to give
\begin{align*}
    &\Egiven{Y_i}{Z_i=1,D_i=d,\vec X_i}
    -
    \Egiven{Y_i}{Z_i=0,D_i=d,\vec X_i}
    \\
    &=
    \Egiven{Y_i(1,d)-Y_i(0,d)}
        {D_i(1)=d,\vec X_i}
    \\
    &\quad+
        \Egiven{Y_i(0,d)}
            {D_i(1)=d,\vec X_i}
        -
        \Egiven{Y_i(0,d)}
            {D_i(0)=d,\vec X_i}.
\end{align*}
The first term is the direct effect with the mediator held at $d$ among individuals for whom $D_i(1)=d$.
The second term is selection bias arising from differences in untreated potential outcomes between individuals for whom $D_i(1)=d$ and individuals for whom $D_i(0)=d$.

To relate the first term to $\operatorname{ADE}(z)$, note that
\[
    \Probgiven{D_i=d}{Z_i=z,\vec X_i}
    =
    \Probgiven{D_i(z)=d}{\vec X_i},
\]

Unfix $d$, to note the target conditional ADE as a weighted average of fixed-mediator direct effects local to the individuals with $D_i(z) = d$:
\begin{align*}
    & \Egiven{
        Y_i(1,D_i(z))-Y_i(0,D_i(z))
    }{\vec X_i}
    \\
    &=
    \sum_{d \in \{0,1\}}
    \Probgiven{D_i=d}{Z_i=z,\vec X_i}
    \Egiven{
        Y_i(1,d)-Y_i(0,d)
    }{
        Z_i=z,D_i = d,\vec X_i
    }
    \\
    &=
    \sum_{d \in \{0,1\}}
    \Probgiven{D_i=d}{Z_i=z,\vec X_i}
    \Egiven{
        Y_i(1,d)-Y_i(0,d)
    }{
        D_i(z)=d,\vec X_i
    }
\end{align*}
where the second equality follows from treatment quasi-random assignment.

Substituting the fixed-$d$ decomposition into the conventional CM estimand gives
\begin{align*}
    &\E[D_i \,\mid\, Z_i=z, \vec X_i]{
        \Egiven{Y_i}{Z_i=1,D_i,\vec X_i} - \Egiven{Y_i}{Z_i=0,D_i,\vec X_i}} \\
    &=\sum_{d \in \{0,1\}} \Probgiven{D_i=d}{Z_i=z,\vec X_i} \Big(
        \Egiven{Y_i}{Z_i=1,D_i=d,\vec X_i} - \Egiven{Y_i}{Z_i=0,D_i=d,\vec X_i} \Big) \\
    &=
    \sum_{d \in \{0,1\}}
    \Probgiven{D_i=d}{Z_i=z,\vec X_i}
    \Egiven{
        Y_i(1,d)-Y_i(0,d)
    }{
        D_i(1)=d,\vec X_i
    }
    \\
    &\quad+
    \sum_{d \in \{0,1\}}
    \Probgiven{D_i=d}{Z_i=z,\vec X_i}
    \Big(
        \Egiven{Y_i(0,d)}
            {D_i(1)=d,\vec X_i}
        -
        \Egiven{Y_i(0,d)}
            {D_i(0)=d,\vec X_i}
    \Big).
\end{align*}

Separate the group difference from selection by add and subtract the conditional ADE, and average over $\vec X_i$ to give the final terms
\begin{align*}
    &\E[\vec X_i]{\E[D_i \,\mid\, Z_i=z, \vec X_i]{
        \Egiven{Y_i}{Z_i=1,D_i,\vec X_i} - \Egiven{Y_i}{Z_i=0,D_i,\vec X_i}}} \\
    &= \text{ADE}(z) \\
    &\quad+
    \underbrace{ \E[\vec X_i]{
        \E[D_i = d\,| \, Z_i = z, \vec X_i]{
            \Egiven{Y_i(0,d)}
                {D_i(1)=d,\,\vec X_i}
            -
            \Egiven{Y_i(0,d)}
                {D_i(0)=d,\,\vec X_i}}}
    }_{
        \text{Selection bias}
    }
    \\
    &\quad+
    \underbrace{
        \E[\vec X_i]{\E[D_i = d \,| \, Z_i = z, \vec X_i]{
            \Egiven{
                Y_i(1,d)-Y_i(0,d)
            }{
                D_i(1)=d,\vec X_i
            }
            -
            \Egiven{
                Y_i(1,d)-Y_i(0,d)
            }{
                D_i(z)=d,\vec X_i
            }}}
    }_{
        \text{Group-difference bias.}
    }
\end{align*}

Note that in the setting $z=1$, the ADE(1) group-difference term equal to zero, group-difference term equal to zero, because both conditional direct effects are evaluated among individuals for whom  $D_i(1)=d$.

\subsubsection{Bias in the Average Indirect Effect (AIE)}
To show that the conventional approach to mediation gives an estimate for the AIE with selection and group difference-bias, start with the definition of the AIE --- the indirect effect among compliers times the size of the complier group.
This sequence relies on assuming no mediator defiers; the following appendix subsection adds another bias term in the case that mediator defiers exist.

Fix $z \in \{0,1\}$.
This proof starts with the conditional AIE, conditional on a specific value of $\vec X_i$.
\begin{align*}
    &\Egiven{ Y_i(z, D_i(1)) - Y_i(z, D_i(0))}{\vec X_i} \\
    &= \Probgiven{D_i(0)=0, D_i(1)=1}{\vec X_i}
        \Egiven{ Y_i(z, 1) - Y_i(z, 0)}{D_i(0)=0, D_i(1)=1, \vec X_i}
\end{align*}
When $D_i$ is not ignorable, the bias comes from attempting to identify the second term,
\\ $\Egiven{ Y_i(z, 1) - Y_i(z, 0)}{D_i(0)=0, D_i(1)=1, \vec X_i}$, the indirect effect among mediator compliers.

Again, note the mean outcomes in terms of average potential outcomes,
\begin{align*}
    \Egiven{Y_i}{Z_i = z, D_i = 1, \vec X_i}
    =& \Egiven{Y_i(z, 1)}{D_i(z) = 1, \vec X_i}, \\
    \Egiven{Y_i}{Z_i = z, D_i = 0, \vec X_i}
    =& \Egiven{Y_i(z, 0)}{D_i(z) = 0, \vec X_i}.
\end{align*}
Compose the selection bias term, as follows.
\begin{align*}
    & \Egiven{Y_i}{Z_i = z, D_i = 1, \vec X_i}
    - \Egiven{Y_i}{Z_i = z, D_i = 0, \vec X_i} \\
    &= \Egiven{Y_i(z, 1)}{D_i(z) = 1, \vec X_i}
        - \Egiven{Y_i(z, 0)}{D_i(z) = 0, \vec X_i} \\
    &= \Egiven{Y_i(z, 1) - Y_i(z, 0)}{D_i(z) = 1, \vec X_i}
        + \Egiven{Y_i(z, 0)}{D_i(z) = 1, \vec X_i}
        - \Egiven{Y_i(z, 0)}{D_i(z) = 0, \vec X_i}
\end{align*}
The right-hand side is the sum of the mediator effect among individuals with $D_i(z)=1$ and a selection-bias term capturing differences in baseline potential outcomes between individuals with $D_i(z)=1$ and $D_i(z)=0$

The AIE is the indirect effect among compliers times the size of the complier group, so we need to compensate for the difference between the med $D_i(z) = 1$ and mediator-complier group $D_i(0)=0, D_i(1)=1$.
\begin{align*}
    &\Egiven{ Y_i(z,1)-Y_i(z,0) }{ D_i(z)=1,\vec X_i } \\
    &= \Egiven{ Y_i(z,1)-Y_i(z,0) }{ D_i(0)=0,D_i(1)=1,\vec X_i } \\
    &\qquad+ \Egiven{ Y_i(z,1)-Y_i(z,0) }{ D_i(z)=1,\vec X_i } - \Egiven{ Y_i(z,1)-Y_i(z,0) }{ D_i(0)=0,D_i(1)=1,\vec X_i }
    .
\end{align*}
The first term is the mediator effect among mediator compliers, while the remaining difference is the group-difference bias from using the mediator effect among individuals with $D_i(z)=1$ in its place.

Collect everything together, as follows.
\begin{align*}
    &\Egiven{Y_i}{Z_i = z, D_i = 1, \vec X_i}
    - \Egiven{Y_i}{Z_i = z, D_i = 0, \vec X_i} \\
    &= \underbrace{
        \Egiven{Y_i(z, 1) - Y_i(z, 0)}{D_i(1) =1, D_i(0)=0,\vec X_i}}_{
            \text{Effect of $D_i$ among compliers, conditional on }\vec X_i} \\
    &+ \underbrace{
        \Egiven{Y_i(z, 0)}{D_i(z) = 1, \vec X_i}
            - \Egiven{Y_i(z, 0)}{D_i(z) = 0, \vec X_i}}_{
                \text{Selection bias}} \\
    &+ \underbrace{\Egiven{Y_i(z, 1) - Y_i(z, 0)}{D_i(z)=1, \vec X_i}
            - \Egiven{Y_i(z, 1) - Y_i(z, 0)}{D_i(0)=0, D_i(1)=1, \vec X_i} }_{
        \text{group difference-bias}}
\end{align*}
The proof is finally achieved by multiplying by the complier score, 
$\Probgiven{D_i(0)=0, D_i(1)=1}{\vec X_i}$
$= \Egiven{D_i}{Z_i = 1, \vec X_i} - \Egiven{D_i}{Z_i = 0, \vec X_i}$,
then averaging across $\vec X_i$.

\subsection{The AIE Without Mediator Monotonicity}
\label{appendix:aie-with-defiers}
The main text maintains mediator monotonicity,
\[
    \Prob{ D_i(0) \leq D_i(1)} = 1.
\]
This assumption rules out mediator defiers and allows the mediator first-stage to be interpreted as the mediator complier share.
This section relaxes mediator monotonicity and shows how the AIE decomposition changes when mediator defiers are present.

For each value of \( \vec X_i\), define the mediator complier and defier shares as
\[
    \pi_C( \vec X_i)
    :=
    \Pr\bigl(D_i(0)=0,D_i(1)=1 \mid \vec X_i\bigr),
\]
and
\[
    \pi_D( \vec X_i)
    :=
    \Pr\bigl(D_i(0)=1,D_i(1)=0 \mid \vec X_i\bigr).
\]
The mediator first-stage is then
\begin{align}
    & \Egiven{D_i }{ Z_i=1, \vec X_i }
    - \Egiven{D_i }{ Z_i=0, \vec X_i }
    \notag\\
    &\qquad
    = \Egiven{D_i(1)-D_i(0)}{ \vec X_i }
    \notag\\
    &\qquad
    =
    \pi_C( \vec X_i)-\pi_D( \vec X_i).
    \label{eq:first-stage-with-defiers}
\end{align}
Thus, absent monotonicity, the mediator first-stage identifies the net complier share rather than the mediator complier share.

Fix $z \in \{0,1\}$. Because \(D_i(0)\) and \(D_i(1)\) are binary,
\[ Y_i\bigl(z,D_i(1)\bigr)
    -
    Y_i\bigl(z,D_i(0)\bigr)
    =
    \bigl[D_i(1)-D_i(0)\bigr]\bigl[Y_i(z, 1) - Y_i(z, 0)\bigr].
\]
Always-takers and never-takers make no contribution to this expression because their mediator status does not respond to \(Z_i\).
Compliers contribute positively, while defiers contribute negatively.

Define the average effects of the mediator among compliers and defiers, respectively, as
\[ \tau_C(z;\vec X_i)
    \coloneqq
    \Egiven{Y_i(z, 1) - Y_i(z, 0)}{ 
        D_i(0)=0,D_i(1)=1, \vec X_i}
\]
\[ \tau_D(z;\vec X_i)
    \coloneqq
    \Egiven{Y_i(z, 1) - Y_i(z, 0)}{ 
        D_i(0)=1,D_i(1)=0, \vec X_i}.
\]
The conditional AIE can therefore be written as
\[ \Egiven{
        Y_i\bigl(z,D_i(1)\bigr)
        -
        Y_i\bigl(z,D_i(0)\bigr)}{ \vec X_i}
    =
    \pi_C( \vec X_i)\tau_C(z;\vec X_i)
    -
    \pi_D( \vec X_i)\tau_D(z;\vec X_i). \]
Applying the expectation across \( \vec X_i\) gives
\begin{align}
    \operatorname{AIE}(z)
    = \E{\pi_C( \vec X_i)\tau_C( z;\vec X_i)
        - \pi_D( \vec X_i)\tau_D( z;\vec X_i) }.
    \label{eq:unconditional-aie-with-defiers}
\end{align}

Using \autoref{eq:first-stage-with-defiers}, this can alternatively be written as
\begin{align}
    &\Egiven{
        Y_i\bigl(z,D_i(1)\bigr)
        -
        Y_i\bigl(z,D_i(0)\bigr)}{ \vec X_i }
    \notag\\
    &\qquad
    =
    \Big(
        \Egiven{D_i}{ Z_i=1, \vec X_i }
        -
        \Egiven{D_i}{ Z_i=0, \vec X_i }
    \Big)
    \tau_C(z;\vec X_i)
    \notag\\
    &\qquad\quad
    +
    \pi_D(\vec X_i)
    \left[
        \tau_C(z; \vec X_i)-\tau_D(z; \vec X_i)
    \right].
    \label{eq:aie-net-first-stage}
\end{align}
The first term is the net mediator first-stage multiplied by the average mediator effect among compliers.
The second term adjusts for differences between the average mediator effects of compliers and defiers.

Under mediator monotonicity, \(\pi_D(\vec X_i)=0\), and equation \eqref{eq:aie-net-first-stage} reduces to the complier-based decomposition used in the main text.
The additional term also vanishes if compliers and defiers have the same average mediator effect, although this is a treatment-effect homogeneity restriction rather than monotonicity.
Without either restriction, the mediator first-stage cannot be interpreted as the mediator complier share, and the AIE cannot be written as the mediator first-stage multiplied only by the average mediator effect among compliers.

The presence of mediator defiers also changes the selection-bias decomposition for the conventional CM estimand. Absent mediator monotonicity, the conventional CM estimand for the AIE can be written as
\begin{align*}
    &\E[\vec X_i]{ \Big( \Egiven{D_i}{Z_i=1,\vec X_i} - \Egiven{D_i}{Z_i=0,\vec X_i} \Big) \Big( \Egiven{Y_i}{Z_i=z,D_i=1,\vec X_i} - \Egiven{Y_i}{Z_i=z,D_i=0,\vec X_i} \Big) } \notag\\
    &= \operatorname{AIE}(z)\\
    &\quad+ \underbrace{ \E[\vec X_i]{ \Big( \pi_C(\vec X_i)-\pi_D(\vec X_i) \Big) \Big( \Egiven{Y_i(z,0)} {D_i(z)=1,\vec X_i} - \Egiven{Y_i(z,0)} {D_i(z)=0,\vec X_i} \Big) } }_{ \text{Selection bias} } \\
    &\quad+ \underbrace{ \E[\vec X_i]{ \Big( \pi_C(\vec X_i)-\pi_D(\vec X_i) \Big) \Big( \Egiven{Y_i(z,1)-Y_i(z,0)} {D_i(z)=1,\vec X_i} - \tau_C(z;\vec X_i) \Big) } }_{ \text{Group-difference bias} } \notag\\
    &\quad+ \underbrace{ \E[\vec X_i]{ \pi_D(\vec X_i) \Big( \tau_D(z;\vec X_i) - \tau_C(z;\vec X_i) \Big) } }_{ \text{Defier compensation} }.
\end{align*}

The second term is selection bias: differences in baseline potential outcomes between those who take and refuse the mediator.
The third term is group-difference bias: the average mediator effect among those observed taking the mediator may differ from the average mediator effect among mediator compliers.
Both terms are weighted by the net mediator first-stage, \(\pi_C(\vec X_i)-\pi_D(\vec X_i)\), rather than the mediator complier share alone.
The final term is the additional defier-compensation term. It arises because the true AIE contains the negatively weighted mediator effect among defiers, while the conventional CM estimand weights its second-stage mediator contrast by the net mediator first-stage.

\subsection{A Regression Framework for Direct and Indirect Effects}
\label{appendix:regression-model}
The first-stage regression of $Z \to Y$ has unbiased estimates, since $Z_i \indep D_i(.) \big| \vec X_i$.
Put $\pi(z; \vec X) = \Egiven{D_i(z)}{\vec X}$, and $\eta_{z, i} = D_i(z) - \pi(z; \vec X)$ the first-stage error terms.
\begin{align*}
    D_i &= Z_i D_i(1) + (1 - Z_i) D_i(0) \\
        &= D_i(0) +
            Z_i \left[ D_i(1) - D_i(0) \right] \\
        &= \underbrace{\pi(0; \vec X_i) 
        }_{\text{Intercept, } \coloneqq \theta + \zeta(\vec X_i)} +
            \underbrace{Z_i \big( \pi(1; \vec X_i) - \pi(0; \vec X_i) \big)}_{
                \text{Regressor, } \coloneqq \bar\pi Z_i }
        + \underbrace{(1- Z_i) \eta_{0,i} + Z_i \eta_{1,i}}_{
            \text{Errors, } \coloneqq \eta_i} \\
    \implies \Egiven{D_i}{Z_i, \vec X_i}
        &= \theta + \bar \pi Z_i + \zeta(\vec X_i).
\end{align*}
Since the quasi-random assignment assumption gives $\Egiven{Z_i \eta_{z,i}}{\vec X_i} = \Egiven{Z_i}{\vec X_i} \Egiven{\eta_{z,i}}{\vec X_i} = 0$, for each $z =0,1$.

The same cannot be said of the regression that estimates direct and indirect effects, without further assumptions.
Write the mean potential outcome as $\mu_{d}(z; \vec X) = \Egiven{Y_i(z, d)}{\vec X}$, with residual $U_i(z,d) = Y_i(z, d) - \mu_{d}(z; \vec X)$ for each $z,d=0,1$.
\begin{align*}
    Y_i &= Z_i Y_i(1, D_i(1)) + (1 - Z_i) Y_i(0, D_i(0)) \\
        &= Z_i D_i Y_i(1, 1) \\
        & \;\;\;\; + (1 - Z_i) D_i Y_i(0, 1) \\
        & \;\;\;\; + Z_i (1 - D_i) Y_i(1, 0) \\
        & \;\;\;\; + (1 - Z_i) (1 - D_i) Y_i(0, 0) \\
        &= Y_i(0, 0) \\
        & \;\;\;\; + Z_i \left[Y_i(1, 0) - Y_i(0, 0) \right] \\
        & \;\;\;\; + D_i \left[Y_i(0, 1) - Y_i(0, 0) \right] \\
        & \;\;\;\; + Z_i D_i \left[Y_i(1, 1) - Y_i(1, 0)
            - \left( Y_i(0, 1) - Y_i(0, 0) \right)\right]
\end{align*}
And so $Y_i$ can be written as a regression equation in terms of the observed factors and error terms.
\begin{align*}
    Y_i &= \mu_0(0; \vec X_i) \\
        & \;\;\;\; + D_i \left[\mu_1(0; \vec X_i) - \mu_0(0; \vec X_i) \right] \\
        & \;\;\;\; + Z_i \left[\mu_0(1; \vec X_i) - \mu_0(0; \vec X_i) \right] \\
        & \;\;\;\; + Z_i D_i \left[\mu_1(1; \vec X_i) - \mu_0(1; \vec X_i)
            - \left( \mu_1(0; \vec X_i) - \mu_0(0; \vec X_i) \right)\right] \\
        & \;\;\;\; + (1-Z_i) \Big[
            \left( 1 - D_i \right) U_i(0, 0) + D_i U_i(0, 1) \Big] \\
        & \;\;\;\; + Z_i \Big[
            \left( 1 - D_i \right) U_i(1, 0) + D_i U_i(1, 1) \Big] \\
        &= \alpha + \beta D_i + \gamma Z_i + \delta Z_i D_i
            + \varphi(\vec X_i) + \varepsilon_i
\end{align*}
With the following definitions:
\begin{enumerate}[label=\textbf{(\alph*)}]
    \item $\alpha = \E{\mu_0(0; \vec X_i)}$ and $\varphi(\vec X_i) = \mu_0(0; \vec X_i) - \alpha$ are the intercept terms.
    \item $\beta = \mu_1(0; \vec X_i) - \mu_0(0; \vec X_i)$ is the indirect effect under $Z_i = 0$
    \item $\gamma = \mu_0(1; \vec X_i) - \mu_0(0; \vec X_i)$ is the direct effect under $D_i = 0$.
    \item $\delta = \mu_1(1; \vec X_i) - \mu_0(1; \vec X_i)- \left( \mu_1(0; \vec X_i) - \mu_0(0; \vec X_i) \right)$ is the interaction effect.
    \item $\varepsilon_i$ is the disruptive error term, with the following definition
    \[ \varepsilon_i = \begin{cases}
        \left(1 - D_i \right) U_i(0,0) + D_i U_i(0,1),& \text{ if } Z_i = 0 \\
        \left(1 - D_i \right) U_i(1,0) + D_i U_i(1,1),& \text{ if } Z_i = 1.
    \end{cases} \]
\end{enumerate}

Taking the conditional expectation, and collecting for the expressions of the ADE and AIE:
\begin{align*}
    \text{ADE(z)}
        &= \E{Y_i(1, D_i(z)) - Y_i(0, D_i(z))}
        \;= \Egiven{\gamma + \delta D_i}{Z_i = z} \\
    \text{AIE(z)}
        &=\E{Y_i(z, D_i(1)) - Y_i(z, D_i(0))}
        \;= \E{\bar \pi \left( \beta + \delta z + \tilde U_i(z) \right)}
\end{align*}
where $\tilde U_i(z) = \Egiven{ U_i(z, 1) - U_i(z, 0)}{D_i(0)=0, D_i(1)=1, \vec X_i}$, for each $z=0,1$.

These equations have simpler expressions after assuming constant treatment effects in a linear framework.
However, unobserved selection-into-$D_i$ based on unobserved costs and benefits only makes sense in the case of heterogeneous treatment effects.
This means a constant treatment effect model is inappropriate for causal mediation analyses of economic settings.

The ultimate regression equation leads to the following expected values, with omitted variables bias coming from the correlated error term, $\varepsilon_i$.
\begin{align*}
    \Egiven{Y_i}{Z_i, D_i, \vec X_i} \;\; =& \;\;
        \alpha
        + \beta D_i
        + \gamma Z_i
        + \delta Z_i D_i
        + \varphi(\vec X_i) \\
        & \;\; \underbrace{\begin{aligned} &+(1-Z_i) \Big[
            \left( 1 - D_i \right) \Egiven{ U_i(0,0) }{D_i(0) = 0, \vec X_i}
            + D_i \Egiven{ U_i(0,1) }{D_i(0) = 1, \vec X_i}\Big] \\
        & + Z_i \Big[
            \left( 1 - D_i \right) \Egiven{ U_i(1,0) }{D_i(1) = 0, \vec X_i}
            + D_i \Egiven{ U_i(1,1) }{D_i(1) = 1, \vec X_i}\Big]
        \end{aligned}}_{
                \text{Omitted variables bias.}
            }
\end{align*}
If $D_i$ was quasi-randomly assigned, then $\Egiven{ U_i(z,d) }{D_i(z) = d, \vec X_i} = 0$; if not, then there terms comprise non-zero omitted variables bias.

These equations have simpler expressions after assuming constant treatment effects in a linear framework;
I have avoided this as having compliers, and controlling for observed factors $\vec X_i$ only makes sense in the case of heterogeneous treatment effects.
These regression terms are conventionally estimated in a simultaneous regression \citep{imai2010identification}.
If sequential quasi-random assignment does not hold, then the regression estimates from estimating the mediation equations (without adjusting for the contaminated bias term) suffer from omitted variables bias.

The following collects these omitted variables bias, omitting the conditional on $\vec X_i$ for brevity.

\makebox[\textwidth]{\parbox{1.25\textwidth}{
\begin{align*}
    \Egiven{Y_i}{Z_i = D_i = 0}
        &= \alpha + \Egiven{ U_i(0,0) }{D_i(0) = 0} \\
    \Egiven{Y_i}{Z_i = 0, D_i = 1}

\subsection{Monotonicity $\implies$ Selection Model, in a CM Setting.}
\label{appendix:mte-monotonicity}
\textit{Proof that (conditional) monotonicity implies a selection model representation in a CM setting.
    This proof is an applied example of the \cite{vytlacil2002independence} equivalence result, now including conditioning covariates $\vec X_i$, and is presented merely as a validation exercise.
}

Assume condition monotonicity \ref{mte:monotonicity} holds, for any treatment values $z < z'$ and any covariate value $\vec X_i = \vec x$.
\[ \Probgiven{D_i(z) \geq D_i(z)}{\vec x} = 1. \]
For each value of $\vec{X}_i = \vec{x}$ and any treatment values $z < z'$, we first define:
\begin{itemize}
    \item $\mathcal{A} = \{i : D_i(z) = D_i(z') = 1\}$, always-mediators
    \item $\mathcal{N} = \{i : D_i(z) = D_i(z') = 0\}$, never-mediators
    \item $\mathcal{C} = \{i : D_i(z) = 0, D_i(z') = 1\}$, mediator-compliers.
\end{itemize}
For any mediator complier $i \in \mathcal C$, partition the set as follows.
\begin{itemize}
    \item $\mathcal{Z}_1(i) = \{z : D_i(z) = 1\}$, treatment values where $i$ takes the mediator
    \item $\mathcal{Z}_0(i) = \{z : D_i(z) = 0\}$, treatment values where $i$ doesn't take the mediator.
\end{itemize}
Note that having binary $Z_i = 0,1$ reduces this to the simple case of $\mathcal{Z}_0(i) = \{0\}$, and $\mathcal{Z}_1(i) = \{1\}$.
The equivalence result holds for continuous values of $Z_i$, so continue with the more general $\mathcal{Z}_0(i), \mathcal{Z}_1(i)$ notation.

By monotonicity, we have
\[ \sup_{z \in \mathcal{Z}_0(i)} \pi(z;\vec{x})
    \leq \inf_{z \in \mathcal{Z}_1(i)} \pi(z;\vec{x}), \;\;
    \text{ for any } i \in \mathcal{C} \]
where $\pi(z;\vec{x}) = \Probgiven{D_i = 1}{Z_i = z, \vec{X}_i = \vec{x}}$ is the mediator propensity score.
A simple proof by contradiction verifies this statement (\citealt[Lemma 1]{vytlacil2002independence}).

Now we construct $V_i$ as follows:
\[ V_i = 
    \begin{cases}
        1, & \text{if } i \in \mathcal{N} \\
        0, & \text{if } i \in \mathcal{A} \\
        \inf_{z \in \mathcal{Z}_1(i)} \pi(z;\vec{x}), & \text{if } i \in \mathcal{C}.
    \end{cases} \]
Define $\psi(z;\vec{x}) = \pi(z;\vec{x})$.
Then we can represent $D_i(z)$ as a selection model,
\[ D_i(z) = \indicator{\psi(z;\vec{X}_i) \geq V_i}, \;\text{ for } z=0,1. \]
We can verify this works:
\begin{itemize}
    \item For $i \in \mathcal{A}$: $V_i = 0$ and $\psi(z;\vec{x}) \geq 0$ for all $z$, so $D_i(z) = 1$
    
    \item For $i \in \mathcal{N}$: $V_i = 1$ and $\psi(z;\vec{x}) \leq 1$ for all $z$, with $\psi(z;\vec{x}) < 1$ for $z \in \mathcal{Z}_0(i)$, so $D_i(z) = 0$ for $z \in \mathcal{Z}_0(i)$
    
    \item For $i \in \mathcal{C}$: $V_i = \inf_{z \in \mathcal{Z}_1(i)} \pi(z;\vec{x})$
    \begin{itemize}
        \item When $z \in \mathcal{Z}_1(i)$: $\psi(z;\vec{x}) \geq \inf_{z' \in \mathcal{Z}_1(i)} \pi(z';\vec{x}) = V_i$, so $D_i(z) = 1$
        \item When $z \in \mathcal{Z}_0(i)$: $\psi(z;\vec{x}) < \inf_{z' \in \mathcal{Z}_1(i)} \pi(z';\vec{x}) = V_i$, so $D_i(z) = 0$.
    \end{itemize}
\end{itemize}
Therefore, the construction $D_i(z) = \indicator{\psi(z;\vec{X}_i) \geq V_i}$ is a valid representation of the selection process under monotonicity.

This selection model can be transformed to one with a uniform distribution, to get the general selection model of \cite{heckman2005structural}.
Let $F_V \big( .\, \big|\, \vec{X}_i \big)$ be the conditional cumulative density function of $V_i$ given $\vec{X}_i$.
Define
\begin{align*}
    U_i &= F_V \left( V_i \mid \vec{X}_i \right) \\
    \pi(z; \vec{X}_i) 
        &= F_V\left( \psi(z; \vec{X}_i) \mid \vec{X}_i \right)
        = \Probgiven{D_i = 1}{Z_i = z, \vec{X}_i}
\end{align*}
We can then equivalently represent the mediator choice as the transformed selection model
\[ D_i(z) = \indicator{\pi(z; \vec{X}_i) \geq U_i}, \;\;\; \text{for } z=0,1 \]
where $U_i \,| \,\vec{X}_i \sim \text{Uniform}(0,1)$ by the probability integral transformation.

\subsection{Mediator MTE Instrument}
\label{appendix:mte-iv}

This subsection states the relevance, exclusion, assignment, overlap, and support conditions imposed on the mediator instrument $\vec X_i^{\mathrm{IV}}$.
Write $\vec X_i=(\vec X_i^{\mathrm{IV}},\vec X_i^-)$, where $\vec X_i^{\mathrm{IV}}$ denotes the variables excluded from the outcome equation and $\vec X_i^-$ contains the remaining controls.
For each treatment arm $z=0,1$, define the mediator propensity score
\[
    \pi\big(z;\vec x^{\mathrm{IV}},\vec x^-\big)
    =
    \Prob{
        D_i=1
        \mid
        Z_i=z,
        \vec X_i^{\mathrm{IV}}=\vec x^{\mathrm{IV}},
        \vec X_i^-=\vec x^-
    }.
\]
Define the realised mediator propensity score as
\[
    P_i
    =
    \pi\big(
        Z_i;
        \vec X_i^{\mathrm{IV}},
        \vec X_i^-
    \big).
\]
For each $z$ and $\vec x^-$, let
\[
    \mathcal P_z(\vec x^-)
    =
    \operatorname{Supp}
    \left(
        P_i
        \mid
        Z_i=z,
        \vec X_i^-=\vec x^-
    \right)
\]
denote the IV-guided propensity-score support within treatment arm $z$.

\begin{enumerate}[label=\textbf{(\arabic*)}]

    \item \emph{Relevance.}
    Conditional on $Z_i=z$ and $\vec X_i^-=\vec x^-$, the mediator propensity score is nonconstant in the excluded mediator instrument:
    \[
        \pi\big(z;\vec x^{\mathrm{IV}},\vec x^-\big)
        \text{ is nonconstant in }
        \vec x^{\mathrm{IV}},
        \qquad
        z=0,1.
    \]
    Equivalently, the excluded instrument generates at least two distinct mediator propensity scores within each treatment arm used for identification:
    \[
        K_z(\vec x^-)
        \coloneqq
        \left|
            \mathcal P_z(\vec x^-)
        \right|
        \geq 2.
    \]

    \item \emph{Exclusion.}
    Conditional on $Z_i$ and $\vec X_i^-$, the excluded mediator instrument affects the outcome only through mediator take-up:
    \[
        Y_i
        \big(
            z,d,\vec x^{\mathrm{IV}}
        \big)
        =
        Y_i(z,d),
    \]
    for all $z,d=0,1$ and all $\vec x^{\mathrm{IV}}$ in the relevant support.

    \item \emph{Conditional quasi-random assignment.}
    Conditional on $Z_i$ and $\vec X_i^-$, the excluded mediator instrument is independent of the latent mediator-selection rank and the potential outcomes:
    \[
        \vec X_i^{\mathrm{IV}}
        \indep
        \left(
            U_i,
            \left\{
                Y_i(z,d):
                z,d\in\{0,1\}
            \right\}
        \right)
        \,\big|\,
        Z_i,\vec X_i^-.
    \]
    Equivalently, variation in $\vec X_i^{\mathrm{IV}}$ changes mediator take-up through the propensity score without independently selecting individuals according to their mediator resistance or potential outcomes.

    \item \emph{Common mediator-selection index.}
    Treatment and the excluded mediator instrument shift mediator take-up through the same scalar latent rank:
    \[
        D_i
        \big(
            z,\vec x^{\mathrm{IV}}
        \big)
        =
        \indicator{
            U_i
            \leq
            \pi
            \big(
                z;
                \vec x^{\mathrm{IV}},
                \vec X_i^-
            \big)
        },
    \]
    for $z=0,1$ and all $\vec x^{\mathrm{IV}}$ in the relevant support, almost surely conditional on $\vec X_i^-$.
    Thus, whenever
    \[
        \pi
        \big(
            z;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big)
        \leq
        \pi
        \big(
            z';
            \vec x^{\mathrm{IV}\prime},
            \vec x^-
        \big),
    \]
    the corresponding mediator-treatment sets are nested:
    \[
        D_i
        \big(
            z,\vec x^{\mathrm{IV}}
        \big)
        \leq
        D_i
        \big(
            z',\vec x^{\mathrm{IV}\prime}
        \big)
        \qquad
        \text{almost surely conditional on }
        \vec X_i^-=\vec x^-.
    \]

    \item \emph{Mediator-cell overlap.}
    At every propensity-score value used to identify the treatment-arm-specific outcome equation, both mediator states occur with positive conditional probability:
    \[
        0
        <
        \Prob{
            D_i=1
            \mid
            Z_i=z,
            P_i=p,
            \vec X_i^-=\vec x^-
        }
        =
        p
        <
        1.
    \]
    Consequently, the mediator-status-specific conditional means
    \[
        \Egiven{
            Y_i
        }{
            Z_i=z,
            D_i=d,
            P_i=p,
            \vec X_i^-=\vec x^-
        }
    \]
    are observed for each $d=0,1$ and each relevant supported value $p\in\mathcal P_z(\vec x^-)$.

    \item \emph{Mediator-complier support.}
    For each $\vec x=(\vec x^{\mathrm{IV}},\vec x^-)$, define the treatment-induced mediator-complier interval as
    \[
        \mathcal C(\vec x)
        =
        \left[
            \pi
            \big(
                0;
                \vec x^{\mathrm{IV}},
                \vec x^-
            \big),
            \pi
            \big(
                1;
                \vec x^{\mathrm{IV}},
                \vec x^-
            \big)
        \right].
    \]
    Under mediator monotonicity, the endpoints are ordered so that
    \[
        \pi
        \big(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big)
        \leq
        \pi
        \big(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big).
    \]
    Each endpoint of $\mathcal C(\vec x)$ must lie in the domain over which the restricted treatment-arm-specific outcome curve is identified for both $z=0,1$.
    More precisely,
    \[
        \mathcal C(\vec x)
        \subseteq
        \mathcal I_z(\vec x^-),
        \qquad
        z=0,1,
    \]
    where $\mathcal I_z(\vec x^-)$ is an interval containing $\mathcal P_z(\vec x^-)$ over which the restricted CF basis introduced in \appendixref{appendix:mte-restricted} is defined and identified.
    If both endpoints belong to $\mathcal P_z(\vec x^-)$, the endpoint conditional means are directly identified in observed data.
    If an endpoint lies between supported propensity-score values, its identification follows from the maintained restricted-CF specification and the full-rank condition stated below.

\end{enumerate}

The first three conditions are the standard relevance, exclusion, and conditional assignment conditions for a valid mediator instrument.
The common-index condition ensures that treatment-induced mediator compliers lie on the same latent selection margin traced by the mediator instrument.
Mediator-cell overlap ensures that the selected outcome means for both mediator states contribute identifying information at the relevant propensity-score values.
Mediator-complier support requires the restricted outcome curve within each treatment arm to cover the propensity-score interval generated by changing treatment from $Z_i=0$ to $Z_i=1$.
The conditions do not require identification at infinity or support over the entire unit interval.

\subsection{Restricted MTE-CF Condition}
\label{appendix:mte-restricted}

Proposition~\ref{proposition:secondstage} uses mediator-status-specific CFs to identify a restricted conditional-outcome curve separately within the $Z_i=0$ and $Z_i=1$ treatment arms.
The finite support of the mediator instrument does not nonparametrically identify an unrestricted CF or the derivative of an unrestricted conditional-outcome function.
Instead, the paper imposes a finite-dimensional restriction on the CFs and uses the mediator-status-specific outcome means at the supported propensity-score values to identify its coefficients.

For each treatment arm $z=0,1$, define the selected CFs
\[
    g_{z,0}
    \big(
        p,\vec x^-
    \big)
    =
    \Egiven{
        U_i(z,0)
    }{
        U_i>p,
        \vec X_i^-=\vec x^-
    },
\]
\[
    g_{z,1}
    \big(
        p,\vec x^-
    \big)
    =
    \Egiven{
        U_i(z,1)
    }{
        U_i\leq p,
        \vec X_i^-=\vec x^-
    }.
\]
These CFs are allowed to differ across treatment arms because the relevant potential-outcome residuals are indexed jointly by $z$ and $d$.

The treatment-arm-specific mediator-status outcome means are
\[
    \begin{aligned}
    &\Egiven{
        Y_i
    }{
        Z_i=z,
        D_i=0,
        P_i=p,
        \vec X_i^-=\vec x^-
    }
    \\
    &\qquad=
    \mu_0
    \big(
        z;\vec x^-
    \big)
    +
    g_{z,0}
    \big(
        p,\vec x^-
    \big),
    \end{aligned}
\]
\[
    \begin{aligned}
    &\Egiven{
        Y_i
    }{
        Z_i=z,
        D_i=1,
        P_i=p,
        \vec X_i^-=\vec x^-
    }
    \\
    &\qquad=
    \mu_1
    \big(
        z;\vec x^-
    \big)
    +
    g_{z,1}
    \big(
        p,\vec x^-
    \big).
    \end{aligned}
\]
Equivalently, the pooled CF-augmented estimating equation is
\[
    \begin{aligned}
    Y_i
    ={}&
    \alpha
    +
    \beta D_i
    +
    \gamma Z_i
    +
    \delta Z_iD_i
    +
    \varphi
    \big(
        \vec X_i^-
    \big)
    \\
    &+
    (1-Z_i)(1-D_i)
    g_{0,0}(P_i)
    +
    (1-Z_i)D_i
    g_{0,1}(P_i)
    \\
    &+
    Z_i(1-D_i)
    g_{1,0}(P_i)
    +
    Z_iD_i
    g_{1,1}(P_i)
    +
    \epsilon_i,
    \end{aligned}
\]
where
\[
    \Egiven{
        \epsilon_i
    }{
        Z_i,
        D_i,
        P_i,
        \vec X_i^-
    }
    =
    0.
\]
This equation is an estimating representation and does not by itself require each of $\alpha,\beta,\gamma,\delta$ to be separately interpreted as an identified CM effect.

\subsubsection{Separability from observed heterogeneity}

The first restriction assumes that, after observed heterogeneity has been absorbed by the potential-outcome means, the shape of selection on latent mediator resistance is common across values of $\vec X_i^-$.
For each $z,d=0,1$, define the rank-specific residual mean
\[
    r_{z,d}
    \big(
        u,\vec x^-
    \big)
    =
    \Egiven{
        U_i(z,d)
    }{
        U_i=u,
        \vec X_i^-=\vec x^-
    }.
\]
The separability restriction is
\[
    r_{z,d}
    \big(
        u,\vec x^-
    \big)
    =
    r_{z,d}(u),
    \qquad
    z,d=0,1.
    \label{eqn:mte-cf-separability}
\]
Equivalently, $\vec X_i^-$ may shift the observed potential-outcome means, but it does not unrestrictedly alter the shape of selection on the unobserved mediator rank.
A linear empirical specification for the observed heterogeneity may be written as
\[
    \mu_d
    \big(
        z;\vec x^-
    \big)
    =
    \alpha
    +
    \beta d
    +
    \gamma z
    +
    \delta zd
    +
    \vec\phi_{z,d}'\vec x^-.
\]

\subsubsection{Finite arm-specific propensity-score support}

For each $z$ and $\vec x^-$, write
\[
    \mathcal P_z(\vec x^-)
    =
    \left\{
        p_{z1}(\vec x^-),
        \ldots,
        p_{zK_z}(\vec x^-)
    \right\},
\]
where
\[
    K_z(\vec x^-)
    =
    \left|
        \mathcal P_z(\vec x^-)
    \right|.
\]
The number $K_z(\vec x^-)$ counts the distinct propensity-score values generated by $\vec X_i^{\mathrm{IV}}$ within treatment arm $z$, rather than the number of raw values taken by the instrument.
The propensity-score support and the CFs are allowed to differ between the $Z_i=0$ and $Z_i=1$ arms.

At every supported propensity score, observed data identify the two selected conditional-outcome means
\[
    h_{z,0}
    \big(
        p_{zk},\vec x^-
    \big)
    =
    \Egiven{
        Y_i
    }{
        Z_i=z,
        D_i=0,
        P_i=p_{zk},
        \vec X_i^-=\vec x^-
    },
\]
and
\[
    h_{z,1}
    \big(
        p_{zk},\vec x^-
    \big)
    =
    \Egiven{
        Y_i
    }{
        Z_i=z,
        D_i=1,
        P_i=p_{zk},
        \vec X_i^-=\vec x^-
    }.
\]
The mediator-cell overlap condition in \appendixref{appendix:mte-iv} ensures that both conditional means are observed at each support point used for identification.

\subsubsection{Restricted CF basis}

For each $z,d=0,1$, let
\[
    \vec b_{z,d}(p)
    =
    \left(
        b_{z,d,1}(p),
        \ldots,
        b_{z,d,L_{z,d}}(p)
    \right)'
\]
denote a vector of known basis functions.
The restricted CF specification is
\[
    g_{z,d}(p)
    =
    c_{z,d}
    +
    \vec b_{z,d}(p)'
    \vec\rho_{z,d},
    \qquad
    p\in\mathcal I_z(\vec x^-).
    \label{eqn:restricted-cf-basis}
\]
The constants $c_{z,d}$ are not separately distinguished from the corresponding potential-outcome means.
Accordingly, impose the normalisation
\[
    c_{z,d}=0,
    \qquad
    z,d=0,1,
\]
or equivalently omit the constant from each CF basis and absorb it into $\mu_d(z;\vec x^-)$.
The number of free basis coefficients must not exceed the identifying variation supplied by the observed mediator-status-specific conditional means.

The basis functions must also be compatible with the truncated-mean interpretation of the CFs.
In particular, the rank-specific residual means satisfy
\[
    \int_0^1
        r_{z,d}(u)
    \,du
    =
    0,
    \qquad
    z,d=0,1,
    \label{eqn:cf-zero-mean}
\]
and the relevant selected residual means are generated by integrating these rank-specific residual means over the selected regions.
For example,
\[
    \Egiven{
        U_i(z,d)
    }{
        U_i\leq p
    }
    =
    \frac{1}{p}
    \int_0^p
        r_{z,d}(u)
    \,du,
\]
and
\[
    \Egiven{
        U_i(z,d)
    }{
        U_i>p
    }
    =
    \frac{1}{1-p}
    \int_p^1
        r_{z,d}(u)
    \,du.
\]
The finite-dimensional basis must preserve these mean-zero and truncated-mean restrictions.
This compatibility requirement prevents the CFs from being arbitrary unrelated functions of the propensity score.

\subsubsection{Full-rank condition}

Collect the free potential-outcome mean coefficients and the CF basis coefficients within treatment arm $z$ in the vector
\[
    \vec\vartheta_z
    =
    \left(
        \vec\theta_{z,0}',
        \vec\theta_{z,1}',
        \vec\rho_{z,0}',
        \vec\rho_{z,1}'
    \right)',
\]
where $\vec\theta_{z,d}$ parameterises the observed component of $\mu_d(z;\vec X_i^-)$.
Stack the identified mediator-status-specific conditional means at the supported propensity-score values in
\[
    \vec h_z(\vec x^-)
    =
    \left(
        h_{z,0}(p_{z1},\vec x^-),
        \ldots,
        h_{z,0}(p_{zK_z},\vec x^-),
        h_{z,1}(p_{z1},\vec x^-),
        \ldots,
        h_{z,1}(p_{zK_z},\vec x^-)
    \right)'.
\]
The restricted outcome equations imply the finite system
\[
    \vec h_z(\vec x^-)
    =
    \mathbf Q_z(\vec x^-)
    \vec\vartheta_z,
    \label{eqn:restricted-cf-system}
\]
where $\mathbf Q_z(\vec x^-)$ contains the observed outcome regressors and the evaluations of the known CF bases at the supported propensity-score values.
Identification requires
\[
    \operatorname{rank}
    \left[
        \mathbf Q_z(\vec x^-)
    \right]
    =
    \dim
    \left(
        \vec\vartheta_z
    \right),
    \qquad
    z=0,1.
    \label{eqn:restricted-cf-rank}
\]
The rank condition requires the supported propensity-score values to generate linearly independent evaluations of the maintained CF basis after accounting for the observed outcome regressors and the imposed normalisations.
Dimension counting alone is insufficient, because distinct basis coefficients may otherwise generate the same conditional means at every supported propensity-score value.

\subsubsection{Restricted conditional-outcome curve}

The conditional outcome mean that averages over realised mediator status is
\[
    m
    \big(
        p;
        z,
        \vec x^-
    \big)
    =
    \Egiven{
        Y_i
    }{
        Z_i=z,
        P_i=p,
        \vec X_i^-=\vec x^-
    }.
\]
Because
\[
    \Egiven{
        D_i
    }{
        Z_i=z,
        P_i=p,
        \vec X_i^-=\vec x^-
    }
    =
    p,
\]
the CF-augmented estimating equation implies
\[
    \begin{aligned}
    m^R
    \big(
        p;
        z,
        \vec x^-
    \big)
    ={}&
    (1-p)
    \mu_0
    \big(
        z;\vec x^-
    \big)
    +
    p
    \mu_1
    \big(
        z;\vec x^-
    \big)
    \\
    &+
    (1-p)
    g_{z,0}(p)
    +
    p
    g_{z,1}(p).
    \end{aligned}
    \label{eqn:restricted-outcome-curve}
\]
At every observed support point,
\[
    m^R
    \big(
        p_{zk};
        z,
        \vec x^-
    \big)
    =
    m
    \big(
        p_{zk};
        z,
        \vec x^-
    \big).
\]
The restricted-CF specification and full-rank condition identify the parameters of $m^R$ from the observed conditional means.
They therefore identify the restricted extension $m^R(p;z,\vec x^-)$ over the interval $\mathcal I_z(\vec x^-)$.
Values of the outcome curve between finite support points are identified within this maintained finite-dimensional class, rather than by nonparametric local differentiation.

\subsection{MTE Identification of the ADE and AIE}
\label{appendix:mte-ade-aie}
\textit{Proof of Theorem \ref{thm:mte-identification}.}
Assume that Assumptions~\ref{mte:monotonicity}, \ref{mte:instrument}, \ref{mte:joint-monotonicity}, and \ref{mte:restricted-cf} hold.

Proposition~\ref{proposition:secondstage} establishes identification of the restricted conditional-outcome curve separately within each treatment arm:
\[
    m^R
    \big(
        p;
        z,
        \vec x^-
    \big)
    =
    \Egiven{
        Y_i
    }{
        Z_i=z,
        P_i=p,
        \vec X_i^-=\vec x^-
    },
\]
for $z=0,1$ and for $p$ in the interval $\mathcal I_z(\vec x^-)$ covered by the restricted CF model.

The superscript $R$ indicates that values of the curve between finitely supported propensity-score values are identified within the maintained restricted-CF class.
At observed propensity-score support points, $m^R$ coincides with the observed conditional-outcome mean.
At other points in $\mathcal I_z(\vec x^-)$, $m^R$ is identified by the finite-dimensional CF restriction and the full-column-rank condition.

\subsubsection{Interpretation of the treatment-arm-specific outcome curve}

Under the common mediator-selection index,
\[
    D_i
    \big(
        z',
        \vec x^{\mathrm{IV}}
    \big)
    =
    \indicator{
        U_i
        \leq
        \pi
        \big(
            z';
            \vec x^{\mathrm{IV}},
            \vec X_i^-
        \big)
    }.
\]

For any fixed treatment state $z$ and propensity-score value $p$, define the counterfactual mediator rule
\[
    D_i(p)
    =
    \indicator{
        U_i\leq p
    }.
\]

Conditional exclusion and quasi-random assignment of the mediator instrument imply
\[
    \begin{aligned}
    m^R
    \big(
        p;
        z,
        \vec x^-
    \big)
    &=
    \Egiven{
        Y_i
        \big(
            z,D_i(p)
        \big)
    }{
        \vec X_i^-=\vec x^-
    }.
    \end{aligned}
    \label{eqn:restricted-outcome-policy-interpretation}
\]

Accordingly, evaluating the restricted outcome curve at the treatment-induced propensity score gives
\[
    \begin{aligned}
    &m^R
    \left(
        \pi
        \big(
            z';
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big);
        z,
        \vec x^-
    \right)
    \\
    &\qquad=
    \Egiven{
        Y_i
        \left(
            z,
            D_i
            \big(
                z',
                \vec x^{\mathrm{IV}}
            \big)
        \right)
    }{
        \vec X_i^-=\vec x^-
    },
    \end{aligned}
    \label{eqn:restricted-outcome-endpoint-interpretation}
\]
for $z,z'=0,1$.

Therefore, the two endpoint values of the $Z_i=z$ outcome curve identify the two relevant nested counterfactual outcome means:
\[
    \begin{aligned}
    &m^R
    \left(
        \pi
        \big(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big);
        z,
        \vec x^-
    \right)
    \\
    &\qquad=
    \Egiven{
        Y_i
        \left(
            z,
            D_i
            \big(
                0,
                \vec x^{\mathrm{IV}}
            \big)
        \right)
    }{
        \vec X_i^-=\vec x^-
    },
    \end{aligned}
\]
and
\[
    \begin{aligned}
    &m^R
    \left(
        \pi
        \big(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big);
        z,
        \vec x^-
    \right)
    \\
    &\qquad=
    \Egiven{
        Y_i
        \left(
            z,
            D_i
            \big(
                1,
                \vec x^{\mathrm{IV}}
            \big)
        \right)
    }{
        \vec X_i^-=\vec x^-
    }.
    \end{aligned}
\]

\subsubsection{Identification of the AIE}

For $z=0,1$, define the conditional indirect effect as
\[
    \begin{aligned}
    \operatorname{AIE}
    \big(
        z;
        \vec x^{\mathrm{IV}},
        \vec x^-
    \big)
    =
    \Egiven{
        Y_i
        \left(
            z,
            D_i
            \big(
                1,
                \vec x^{\mathrm{IV}}
            \big)
        \right)
        -
        Y_i
        \left(
            z,
            D_i
            \big(
                0,
                \vec x^{\mathrm{IV}}
            \big)
        \right)
    }{
        \vec X_i^-=\vec x^-
    }.
    \end{aligned}
    \label{eqn:conditional-aie-definition}
\]

Using \eqref{eqn:restricted-outcome-endpoint-interpretation},
\[
    \begin{aligned}
    &\operatorname{AIE}
    \big(
        z;
        \vec x^{\mathrm{IV}},
        \vec x^-
    \big)
    \\
    &\qquad=
    m^R
    \left(
        \pi
        \big(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big);
        z,
        \vec x^-
    \right)
    \\
    &\qquad\quad-
    m^R
    \left(
        \pi
        \big(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big);
        z,
        \vec x^-
    \right).
    \end{aligned}
    \label{eqn:conditional-aie-endpoint-difference}
\]

The mediator-complier support condition ensures that both propensity-score endpoints lie in the domain $\mathcal I_z(\vec x^-)$ over which $m^R$ is identified.

Therefore, the right-hand side of \eqref{eqn:conditional-aie-endpoint-difference} is identified for each $z=0,1$ and each relevant covariate value.

To connect this endpoint expression to the mediator MTE, note that mediator monotonicity and the common-index representation imply
\[
    \begin{aligned}
    &D_i
    \big(
        1,
        \vec x^{\mathrm{IV}}
    \big)
    -
    D_i
    \big(
        0,
        \vec x^{\mathrm{IV}}
    \big)
    \\
    &\qquad=
    \indicator{
        \pi
        \big(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big)
        <
        U_i
        \leq
        \pi
        \big(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big)
    }.
    \end{aligned}
    \label{eqn:mediator-complier-indicator}
\]

For binary mediator values,
\[
    \begin{aligned}
    &Y_i
    \left(
        z,
        D_i
        \big(
            1,
            \vec x^{\mathrm{IV}}
        \big)
    \right)
    -
    Y_i
    \left(
        z,
        D_i
        \big(
            0,
            \vec x^{\mathrm{IV}}
        \big)
    \right)
    \\
    &\qquad=
    \left[
        D_i
        \big(
            1,
            \vec x^{\mathrm{IV}}
        \big)
        -
        D_i
        \big(
            0,
            \vec x^{\mathrm{IV}}
        \big)
    \right]
    \left[
        Y_i(z,1)-Y_i(z,0)
    \right].
    \end{aligned}
\]

Substituting \eqref{eqn:mediator-complier-indicator} gives
\[
    \begin{aligned}
    &Y_i
    \left(
        z,
        D_i
        \big(
            1,
            \vec x^{\mathrm{IV}}
        \big)
    \right)
    -
    Y_i
    \left(
        z,
        D_i
        \big(
            0,
            \vec x^{\mathrm{IV}}
        \big)
    \right)
    \\
    &\qquad=
    \indicator{
        \pi
        \big(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big)
        <
        U_i
        \leq
        \pi
        \big(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big)
    }
    \left[
        Y_i(z,1)-Y_i(z,0)
    \right].
    \end{aligned}
\]

Taking conditional expectations and iterating over the latent mediator rank yields
\[
    \begin{aligned}
    &\operatorname{AIE}
    \big(
        z;
        \vec x^{\mathrm{IV}},
        \vec x^-
    \big)
    \\
    &\qquad=
    \int_{
        \pi(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        )
    }^{
        \pi(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        )
    }
        \Egiven{
            Y_i(z,1)-Y_i(z,0)
        }{
            U_i=p,
            \vec X_i^-=\vec x^-
        }
    \,dp.
    \end{aligned}
\]

By the definition of the mediator MTE,
\[
    \begin{aligned}
    &\operatorname{AIE}
    \big(
        z;
        \vec x^{\mathrm{IV}},
        \vec x^-
    \big)
    \\
    &\qquad=
    \int_{
        \pi(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        )
    }^{
        \pi(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        )
    }
        \operatorname{MTE}
        \big(
            p;
            z,
            \vec x^-
        \big)
    \,dp.
    \end{aligned}
    \label{eqn:conditional-aie-mte-integral}
\]

Within the restricted-CF model, the identified mediator MTE is the restricted mediator MTE,
\[
    \operatorname{MTE}^R
    \big(
        p;
        z,
        \vec x^-
    \big)
    =
    \partialdiff{p}
    m^R
    \big(
        p;
        z,
        \vec x^-
    \big),
\]
for almost every $p$ in $\mathcal I_z(\vec x^-)$.

Absolute continuity of $m^R$ therefore gives
\[
    \begin{aligned}
    &\int_{
        \pi(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        )
    }^{
        \pi(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        )
    }
        \operatorname{MTE}^R
        \big(
            p;
            z,
            \vec x^-
        \big)
    \,dp
    \\
    &\qquad=
    m^R
    \left(
        \pi
        \big(
            1;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big);
        z,
        \vec x^-
    \right)
    \\
    &\qquad\quad-
    m^R
    \left(
        \pi
        \big(
            0;
            \vec x^{\mathrm{IV}},
            \vec x^-
        \big);
        z,
        \vec x^-
    \right).
    \end{aligned}
    \label{eqn:restricted-aie-equivalence}
\]

Thus, the MTE integral and endpoint-difference representations identify the same conditional indirect effect within the maintained restricted model.

Averaging over the joint distribution of $\vec X_i^{\mathrm{IV}}$ and $\vec X_i^-$ identifies
\[
    \begin{aligned}
    \operatorname{AIE}(z)
    =
    \mathbb E
    \Big[
        &
        m^R
        \left(
            \pi
            \big(
                1;
                \vec X_i^{\mathrm{IV}},
                \vec X_i^-
            \big);
            z,
            \vec X_i^-
        \right)
        \\
        &-
        m^R
        \left(
            \pi
            \big(
                0;
                \vec X_i^{\mathrm{IV}},
                \vec X_i^-
            \big);
            z,
            \vec X_i^-
        \right)
    \Big].
    \end{aligned}
    \label{eqn:aie-endpoint-identification}
\]

Equivalently,
\[
    \begin{aligned}
    \operatorname{AIE}(z)
    =
    \E{
        \int_{
            \pi(
                0;
                \vec X_i^{\mathrm{IV}},
                \vec X_i^-
            )
        }^{
            \pi(
                1;
                \vec X_i^{\mathrm{IV}},
                \vec X_i^-
            )
        }
            \operatorname{MTE}^R
            \big(
                u;
                z,
                \vec X_i^-
            \big)
        \,du }.
    \end{aligned}
    \label{eqn:aie-mte-identification}
\]

This establishes identification of $\operatorname{AIE}(z)$ for each $z=0,1$.

\subsubsection{Identification of the ADE}

For $z=0,1$, recall the ADE definition,
\[
    \operatorname{ADE}(z)
    =
    \E{Y_i
        \big(
            1,D_i(z)
        \big)
        -
        Y_i
        \big(
            0,D_i(z)
        \big)}.
    \label{eqn:ade-t-definition}
\]

The total treatment effect is
\[
    \operatorname{ATE}
    =
    \E{
        Y_i
        \big(
            1,D_i(1)
        \big)
        -
        Y_i
        \big(
            0,D_i(0)
        \big)}.
    \label{eqn:ate-nested-definition}
\]

First add and subtract $Y_i(0,D_i(1))$:
\[
    \begin{aligned}
    \operatorname{ATE}
    ={}&
    \E{
        Y_i
        \big(
            1,D_i(1)
        \big)
        -
        Y_i
        \big(
            0,D_i(1)
        \big)}
    \\
    &+
    \E{
        Y_i
        \big(
            0,D_i(1)
        \big)
        -
        Y_i
        \big(
            0,D_i(0)
        \big)}.
    \end{aligned}
\]

The first term is $\operatorname{ADE}(1)$ and the second is $\operatorname{AIE}(0)$.

Therefore,
\[
    \operatorname{ATE}
    =
    \operatorname{ADE}(1)
    +
    \operatorname{AIE}(0).
    \label{eqn:ate-decomposition-z0}
\]

It follows that
\[
    \operatorname{ADE}(1)
    =
    \operatorname{ATE}
    -
    \operatorname{AIE}(0).
    \label{eqn:ade1-identification}
\]

Alternatively, AIE(1) and ADE(0) are identified by the same process, identifying AIE(1) via the Mediator MTE among the $Z_i =1$ sample and using ADE(0) as the ATE residual.
This proves Theorem~\ref{thm:mte-identification}.

\subsection{Semi-parametric Estimation of the AIE and ADE}
\label{appendix:semi-parametric}

This subsection describes the semi-parametric estimation of the AIE and ADE.
For notational convenience, the exposition focuses on the $Z_i=0$ treatment arm, which identifies $\operatorname{AIE}(0)$ and, together with the ATE, $\operatorname{ADE}(1)$.
The estimators of $\operatorname{AIE}(1)$ and $\operatorname{ADE}(0)$ follow symmetrically by applying the same procedure to the $Z_i=1$ treatment arm.

Let
\[
    n_0
    \coloneqq
    \sum_{i=1}^{n}
        \indicator{Z_i=0}
\]
denote the number of observations assigned to the $Z_i=0$ arm.

\subsubsection{Estimation of the Mediator Propensity Score}

The first stage estimates the mediator propensity score,
\[
    \pi
    \big(
        z;
        \vec x^{\mathrm{IV}},
        \vec x^-
    \big)
    =
    \Probgiven{
        D_i=1
    }{
        Z_i=z,
        \vec X_i^{\mathrm{IV}}=\vec x^{\mathrm{IV}},
        \vec X_i^-=\vec x^-
    }.
\]
Let
\[
    \widehat\pi
    \big(
        z;
        \vec x^{\mathrm{IV}},
        \vec x^-
    \big)
\]
denote its estimate, and define the estimated realised propensity score as
\[
    \widehat P_i
    =
    \widehat\pi
    \big(
        Z_i;
        \vec X_i^{\mathrm{IV}},
        \vec X_i^-
    \big).
\]

The propensity score may be estimated using a correctly specified binary-response model or an appropriate semi-parametric estimator.
The estimator must consistently estimate both the realised propensity score and the two counterfactual endpoint propensities,
\[
    \widehat p_{0i}
    \coloneqq
    \widehat\pi
    \big(
        0;
        \vec X_i^{\mathrm{IV}},
        \vec X_i^-
    \big)
\]
and
\[
    \widehat p_{1i}
    \coloneqq
    \widehat\pi
    \big(
        1;
        \vec X_i^{\mathrm{IV}},
        \vec X_i^-
    \big).
\]

The mediator-complier support condition requires both endpoint propensities to lie in the domain identified by mediator-instrument variation within the treatment arm used to estimate the outcome curve.
For the $Z_i=0$ estimator, this requires
\[
    \left[
        \pi
        \big(
            0;
            \vec X_i^{\mathrm{IV}},
            \vec X_i^-
        \big),
        \pi
        \big(
            1;
            \vec X_i^{\mathrm{IV}},
            \vec X_i^-
        \big)
    \right]
    \subseteq
    \mathcal I_0
    \big(
        \vec X_i^-
    \big)
\]
almost surely, where $\mathcal I_0(\vec X_i^-)$ is the domain over which the restricted $Z_i=0$ outcome curve is identified.

\subsubsection{Estimation of the Restricted Outcome Curve}
\label{appendix:mte-estimation}

Within the $Z_i=0$ treatment arm, define
\[
    m
    \big(
        p;
        0,
        \vec x^-
    \big)
    =
    \Egiven{
        Y_i
    }{
        Z_i=0,
        P_i=p,
        \vec X_i^-=\vec x^-
    }.
\]

Let
\[
    \vec b_{0,K}(p)
    =
    \left(
        b_{01}(p),
        \ldots,
        b_{0K}(p)
    \right)'
\]
be a vector of known polynomial, regression-spline, or other series basis functions.
The restricted outcome curve is approximated by
\[
    m_K^R
    \big(
        p;
        0,
        \vec x^-
    \big)
    =
    \vec\phi_0'\vec x^-
    +
    \vec b_{0,K}(p)'\vec\rho_0.
    \label{eqn:restricted-outcome-estimation-model}
\]

An intercept is included either in $\vec x^-$ or in $\vec b_{0,K}(p)$, but not in both.
More generally, the basis must not duplicate terms already included in the linear control function.
The rank condition in \appendixref{appendix:mte-restricted} requires the IV-guided propensity-score variation to distinguish the maintained basis terms from one another and from the included controls.

The coefficients are estimated in the $Z_i=0$ subsample by
\[
    \begin{aligned}
    \left(
        \widehat{\vec\phi}_0,
        \widehat{\vec\rho}_0
    \right)
    =
    \arg \min_{
        \vec\phi,
        \vec\rho
    }
    \frac{1}{n_0}
    \sum_{i:Z_i=0}
    \left[
        Y_i
        -
        \vec\phi'\vec X_i^-
        -
        \vec b_{0,K}
        \big(
            \widehat P_i
        \big)'
        \vec\rho
    \right]^2.
    \end{aligned}
    \label{eqn:restricted-outcome-estimator}
\]

The resulting fitted outcome curve is
\[
    \widehat m
    \big(
        p;
        0,
        \vec x^-
    \big)
    =
    \widehat{\vec\phi}_0'\vec x^-
    +
    \vec b_{0,K}(p)'
    \widehat{\vec\rho}_0.
    \label{eqn:fitted-restricted-outcome-curve}
\]

When the maintained restricted outcome model is exactly finite dimensional, $K$ is fixed.
When the basis is used as a sieve approximation to a smooth outcome curve, $K=K_{n_0}$ may increase with the treatment-arm sample size.
The basis dimension must increase sufficiently quickly to control approximation error, but sufficiently slowly to control estimation error \citep{newey1997convergence,chen2007large}.

A parametric assumption on the joint distribution of the unobserved mediator-selection and outcome errors provides an alternative specification of the propensity-score component of the outcome curve.
For example, joint normality implies the familiar inverse-Mills-ratio correction.
The semi-parametric estimator instead uses a polynomial, spline, or sieve approximation and does not require a parametric distribution for the unobserved errors.

\subsubsection{Estimation of the AIE and ADE}

For each observation in the $Z_i=0$ subsample, evaluate the fitted $Z_i=0$ outcome curve at the mediator propensities induced by both treatment states:
\[
    \widehat m
    \big(
        \widehat p_{0i};
        0,
        \vec X_i^-
    \big)
\]
and
\[
    \widehat m
    \big(
        \widehat p_{1i};
        0,
        \vec X_i^-
    \big).
\]

The fitted conditional indirect effect is
\[
    \begin{aligned}
    \widehat\tau_i^{\,\operatorname{AIE}(0)}
    =
    {}&
    \widehat m
    \big(
        \widehat p_{1i};
        0,
        \vec X_i^-
    \big)
    \\
    &-
    \widehat m
    \big(
        \widehat p_{0i};
        0,
        \vec X_i^-
    \big).
    \end{aligned}
    \label{eqn:conditional-aie-estimator}
\]

Under the partially linear specification in
\eqref{eqn:fitted-restricted-outcome-curve}, the additive control-variable component cancels:
\[
    \begin{aligned}
    \widehat\tau_i^{\,\operatorname{AIE}(0)}
    =
    \left[
        \vec b_{0,K}
        \big(
            \widehat p_{1i}
        \big)
        -
        \vec b_{0,K}
        \big(
            \widehat p_{0i}
        \big)
    \right]'
    \widehat{\vec\rho}_0.
    \end{aligned}
    \label{eqn:conditional-aie-series-estimator}
\]

The cancellation of the additive control term does not make the controls redundant.
They remain necessary for conditional instrument validity, consistent propensity-score estimation, and the separability condition used to identify the restricted outcome curve.

The indirect effect is estimated by averaging these fitted endpoint differences over the $Z_i=0$ subsample:
\[
    \boxed{
    \widehat{\operatorname{AIE}}_0(0)
    =
    \frac{1}{n_0}
    \sum_{i:Z_i=0}
        \widehat\tau_i^{\,\operatorname{AIE}(0)}
    }.
    \label{eqn:aie0-subsample-estimator}
\]

Equivalently,
\[
    \begin{aligned}
    \widehat{\operatorname{AIE}}_0(0)
    =
    \frac{1}{n_0}
    \sum_{i:Z_i=0}
    \Big[
        &
        \widehat m
        \big(
            \widehat p_{1i};
            0,
            \vec X_i^-
        \big)
        \\
        &-
        \widehat m
        \big(
            \widehat p_{0i};
            0,
            \vec X_i^-
        \big)
    \Big].
    \end{aligned}
\]

Quasi-random assignment of $Z_i$ implies that the distribution of
\[
    \left(
        \vec X_i^{\mathrm{IV}},
        \vec X_i^-
    \right)
\]
within the $Z_i=0$ subsample consistently estimates the corresponding target-population distribution.
The subsample average therefore standardises the conditional endpoint differences to the target population.

An alternative estimator standardises the fitted effects over the full sample:
\[
    \begin{aligned}
    \widehat{\operatorname{AIE}}_{\mathrm{std}}(0)
    =
    \frac{1}{n}
    \sum_{i=1}^{n}
    \Big[
        &
        \widehat m
        \big(
            \widehat p_{1i};
            0,
            \vec X_i^-
        \big)
        \\
        &-
        \widehat m
        \big(
            \widehat p_{0i};
            0,
            \vec X_i^-
        \big)
    \Big].
    \end{aligned}
    \label{eqn:aie0-full-standardisation}
\]
Under quasi-random treatment assignment, the subsample and full-sample standardisation estimators have the same probability limit, although their finite-sample and asymptotic variances may differ.

The complementary direct effect is estimated using
\[
    \operatorname{ATE}
    =
    \operatorname{AIE}(0)
    +
    \operatorname{ADE}(1).
\]
Thus,
\[
    \boxed{
    \widehat{\operatorname{ADE}}(1)
    =
    \widehat{\operatorname{ATE}}
    -
    \widehat{\operatorname{AIE}}(0)
    }.
    \label{eqn:ade1-estimator}
\]

Repeating the outcome-curve estimation in the $Z_i=1$ subsample gives
\[
    \widehat{\operatorname{AIE}}(1)
\]
and
\[
    \boxed{
    \widehat{\operatorname{ADE}}(0)
    =
    \widehat{\operatorname{ATE}}
    -
    \widehat{\operatorname{AIE}}(1)
    }.
\]

The endpoint estimator does not require numerical differentiation.
Under absolute continuity,
\[
    \begin{aligned}
    &m^R
    \big(
        p_1;
        0,
        \vec x^-
    \big)
    -
    m^R
    \big(
        p_0;
        0,
        \vec x^-
    \big)
    \\
    &\qquad=
    \int_{p_0}^{p_1}
        \operatorname{MTE}^R
        \big(
            u;
            0,
            \vec x^-
        \big)
    \,du.
    \end{aligned}
\]
The AIE can therefore be estimated directly from the two fitted endpoint values.
This avoids the instability associated with numerically differentiating a flexible estimate of the conditional-outcome curve.

Prediction at
\[
    \widehat\pi
    \big(
        1;
        \vec X_i^{\mathrm{IV}},
        \vec X_i^-
    \big)
\]
is not extrapolation merely because the outcome curve is estimated in the $Z_i=0$ subsample.
It is interpolation when both treatment-induced propensity scores lie in the domain $\mathcal I_0(\vec X_i^-)$ identified by mediator-instrument variation and the restricted-CF model within the $Z_i=0$ arm.
If an endpoint lies outside that domain, its evaluation instead requires unsupported extrapolation.

\subsubsection{Consistency and Asymptotic Normality}

The large-sample argument combines standard results for series or sieve estimation with results for multi-step semi-parametric estimators containing generated regressors.
Series estimators consistently approximate conditional expectations under smoothness, moment, support, approximation, and basis-growth conditions.
Although the estimated outcome curve may converge at a nonparametric rate, sufficiently smooth scalar functionals of the curve can be root-$n$ consistent and asymptotically normal \citep{newey1997convergence,chen2007large}.

Assume that
\[
    q_0
    \coloneqq
    \Prob{Z_i=0}
\]
satisfies
\[
    0<q_0<1.
\]
Then
\[
    \frac{n_0}{n}
    \overset{p}{\longrightarrow}
    q_0.
    \label{eqn:z0-sample-share}
\]

In addition to the identification assumptions, impose the following regularity conditions:

\begin{enumerate}[label=\textbf{(\arabic*)}]

    \item
    The observations are independently and identically distributed, or satisfy an appropriate weak-dependence condition.

    \item
    The outcome has sufficiently many finite moments.
    For some $\epsilon>0$,
    \[
        \E{
            \left|
                Y_i
            \right|^{2+\epsilon}
        }
        <
        \infty.
    \]

    \item
    The mediator propensities are bounded away from zero and one over the target population:
    \[
        0
        <
        \underline p
        \leq
        \pi
        \big(
            z;
            \vec X_i^{\mathrm{IV}},
            \vec X_i^-
        \big)
        \leq
        \overline p
        <
        1,
        \qquad
        z=0,1.
    \]

    \item
    Both treatment-induced propensity-score endpoints lie in the domain of the restricted outcome curve:
    \[
        \left[
            \pi
            \big(
                0;
                \vec X_i^{\mathrm{IV}},
                \vec X_i^-
            \big),
            \pi
            \big(
                1;
                \vec X_i^{\mathrm{IV}},
                \vec X_i^-
            \big)
        \right]
        \subseteq
        \mathcal I_0
        \big(
            \vec X_i^-
        \big)
    \]
    almost surely.

    \item
    The population second-moment matrix of the outcome-curve regressors is nonsingular, with its relevant eigenvalues bounded away from zero and infinity.

    \item
    The restricted outcome curve is sufficiently smooth and can be approximated by the chosen basis with error
    \[
        a_K
        \coloneqq
        \sup_{
            p,\vec x^-
        }
        \left|
            m^R
            \big(
                p;
                0,
                \vec x^-
            \big)
            -
            m_K^R
            \big(
                p;
                0,
                \vec x^-
            \big)
        \right|
        \longrightarrow
        0.
    \]

    \item
    If the sieve dimension increases, then
    \[
        K=K_{n_0}
        \longrightarrow
        \infty
    \]
    sufficiently slowly for sampling error to vanish and sufficiently quickly for the approximation bias in the endpoint functional to be asymptotically negligible.

    \item
    The first-stage estimator is uniformly consistent over the relevant support:
    \[
        \sup_{
            z,\vec x^{\mathrm{IV}},\vec x^-
        }
        \left|
            \widehat\pi
            \big(
                z;
                \vec x^{\mathrm{IV}},
                \vec x^-
            \big)
            -
            \pi
            \big(
                z;
                \vec x^{\mathrm{IV}},
                \vec x^-
            \big)
        \right|
        \overset{p}{\longrightarrow}
        0.
    \]

    \item
    For asymptotic normality, the first-stage estimator possesses an asymptotically linear representation, and the map from the first-stage propensity score and restricted outcome curve to the averaged endpoint difference is sufficiently smooth.

\end{enumerate}

Under these conditions,
\[
    \sup_{
        p,\vec x^-
    }
    \left|
        \widehat m
        \big(
            p;
            0,
            \vec x^-
        \big)
        -
        m^R
        \big(
            p;
            0,
            \vec x^-
        \big)
    \right|
    \overset{p}{\longrightarrow}
    0
    \label{eqn:uniform-outcome-curve-consistency}
\]
over the mediator-complier domain.

Uniform consistency of the fitted curve and consistency of the estimated endpoint propensities imply
\[
    \begin{aligned}
    &
    \widehat m
    \big(
        \widehat p_{1i};
        0,
        \vec X_i^-
    \big)
    -
    \widehat m
    \big(
        \widehat p_{0i};
        0,
        \vec X_i^-
    \big)
    \\
    &\qquad-
    \left[
        m^R
        \big(
            p_{1i};
            0,
            \vec X_i^-
        \big)
        -
        m^R
        \big(
            p_{0i};
            0,
            \vec X_i^-
        \big)
    \right]
    \overset{p}{\longrightarrow}
    0.
    \end{aligned}
\]

The law of large numbers therefore gives
\[
    \widehat{\operatorname{AIE}}_0(0)
    \overset{p}{\longrightarrow}
    \operatorname{AIE}(0).
    \label{eqn:aie0-estimator-consistency}
\]

If
\[
    \widehat{\operatorname{ATE}}
    \overset{p}{\longrightarrow}
    \operatorname{ATE},
\]
then
\[
    \widehat{\operatorname{ADE}}(1)
    \overset{p}{\longrightarrow}
    \operatorname{ADE}(1).
    \label{eqn:ade1-estimator-consistency}
\]

The entire estimated outcome curve need not attain a parametric convergence rate.
The AIE, however, is a scalar functional of the propensity-score function, the restricted outcome curve, and the covariate distribution:
\[
    \begin{aligned}
    \mathcal T_0
    \big(
        m,\pi,F_0
    \big)
    =
    \int
    \Big[
        &
        m
        \big(
            \pi(1;\vec x);
            0,
            \vec x^-
        \big)
        \\
        &-
        m
        \big(
            \pi(0;\vec x);
            0,
            \vec x^-
        \big)
    \Big]
    dF_0(\vec x),
    \end{aligned}
\]
where $F_0$ is the covariate distribution in the $Z_i=0$ arm.

Under sufficient smoothness, a negligible sieve-approximation bias, and appropriate regularity of the generated propensity score, this functional admits an asymptotically linear representation:
\[
    \widehat{\operatorname{AIE}}_0(0)
    -
    \operatorname{AIE}(0)
    =
    \frac{1}{n_0}
    \sum_{i:Z_i=0}
        \psi_{0i}
    +
    o_p
    \left(
        n_0^{-1/2}
    \right).
    \label{eqn:aie0-asymptotic-linearity}
\]

The influence function $\psi_{0i}$ contains contributions from:

\begin{enumerate}[label=\textnormal{(\roman*)}]
    \item sampling variation in the covariate distribution;
    \item estimation of the restricted outcome curve;
    \item estimation of the propensity score used as a generated regressor;
    \item estimation of the two endpoint propensity scores.
\end{enumerate}

The estimated propensity score has a dual role: it is both a conditioning variable in the restricted outcome regression and the argument at which the fitted outcome curve is evaluated.
The contribution of both roles must be included in the asymptotic variance \citep{hahn2013asymptotic}.

It follows that
\[
    \sqrt{n_0}
    \left[
        \widehat{\operatorname{AIE}}_0(0)
        -
        \operatorname{AIE}(0)
    \right]
    \overset{d}{\longrightarrow}
    \mathcal N
    \left(
        0,
        V_{\operatorname{AIE}(0)}
    \right),
    \label{eqn:aie0-root-n0-normality}
\]
where
\[
    V_{\operatorname{AIE}(0)}
    =
    \Egiven{
        \psi_{0i}^2
    }{
        Z_i=0
    }.
\]

Because
\[
    \frac{n_0}{n}
    \overset{p}{\longrightarrow}
    q_0,
\]
the equivalent full-sample normalisation is
\[
    \sqrt n
    \left[
        \widehat{\operatorname{AIE}}_0(0)
        -
        \operatorname{AIE}(0)
    \right]
    \overset{d}{\longrightarrow}
    \mathcal N
    \left(
        0,
        \frac{
            V_{\operatorname{AIE}(0)}
        }{
            q_0
        }
    \right).
    \label{eqn:aie0-root-n-normality}
\]

Thus, use of the $Z_i=0$ subsample affects the asymptotic variance through the treatment-arm probability but does not change the convergence order when $q_0$ is bounded away from zero.

If the ATE estimator is also asymptotically linear, then
\[
    \begin{aligned}
    \widehat{\operatorname{ADE}}(1)
    -
    \operatorname{ADE}(1)
    ={}&
    \left[
        \widehat{\operatorname{ATE}}
        -
        \operatorname{ATE}
    \right]
    \\
    &-
    \left[
        \widehat{\operatorname{AIE}}(0)
        -
        \operatorname{AIE}(0)
    \right].
    \end{aligned}
\]
The ADE estimator is therefore asymptotically normal, with variance accounting for the sampling variance and covariance of the estimated ATE and AIE.

\subsubsection{Bootstrap Inference}

Inference is implemented using a nonparametric bootstrap that repeats the complete estimation procedure.
For each bootstrap replication:

\begin{enumerate}[label=\textbf{(\arabic*)}]

    \item
    Resample observations from the original sample.

    \item
    Re-estimate the mediator propensity-score model.

    \item
    Reconstruct the realised propensity scores and the treatment-induced endpoint propensities.

    \item
    Re-estimate the restricted conditional-outcome curve using the bootstrap $Z_i=0$ subsample.

    \item
    Evaluate the fitted outcome curve at both endpoint propensity scores.

    \item
    Average the endpoint differences to estimate $\operatorname{AIE}(0)$.

    \item
    Re-estimate the ATE.

    \item
    Construct
    \[
        \widehat{\operatorname{ADE}}(1)
        =
        \widehat{\operatorname{ATE}}
        -
        \widehat{\operatorname{AIE}}(0).
    \]

\end{enumerate}

Repeating every stage accounts for uncertainty from propensity-score estimation, estimation of the restricted outcome curve, endpoint evaluation, standardisation over the covariate distribution, and estimation of the ATE.
Under the regularity conditions supporting the asymptotic linear representation, the bootstrap consistently estimates the sampling distributions of the AIE and ADE estimators.

The same procedure applies to estimation of the $Z_i=1$ outcome curve and the resulting estimators of $\operatorname{AIE}(1)$ and $\operatorname{ADE}(0)$.
